</tool>
- append file: 
<tool name="append_file" path="ejemplo.txt">

<tool name="run_python_file" path="example.py"></tool>

<tool name="grep" path="example.py" pattern="contenido a buscar"></tool>

<tool name="get_article_wikipedia" article="article_name"></tool>
